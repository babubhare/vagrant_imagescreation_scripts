#include <stdio.h>

int main() {
    printf("Hello from AArch64!\n");
    printf("If you are reading this, your cross-compiler and QEMU setup are working perfectly.\n");
    return 0;
}

aarch64-unknown-linux-gnu-gcc hello.c -o hello_aarch64
file hello_aarch64

sudo apt update
sudo apt install qemu-user

aarch64-unknown-linux-gnu-gcc -static hello.c -o hello_static
qemu-aarch64 ./hello_static
